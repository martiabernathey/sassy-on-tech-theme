{{- /* Returns the first image URL for a post: photos param, image param, then first in the content. "" if none. */ -}} {{- $img := "" -}} {{- with .Params.photos }}{{ $img = index . 0 }}{{ end -}} {{- if not $img }}{{ with .Params.image }}{{ $img = . }}{{ end }}{{ end -}} {{- if not $img -}} {{- $found := findRE `]+src=["'][^"']+["']` (string .Content) 1 -}} {{- with $found -}} {{- $img = replaceRE `^.*src=["']([^"']+)["'].*$` "$1" (index . 0) -}} {{- end -}} {{- end -}} {{- return $img -}}
